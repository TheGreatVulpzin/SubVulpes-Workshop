# Based on the SDK template. The same target works in the repository and with an extracted SDK.
cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(sv_jukebox C)
    if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "Use MSVC x86 for the jukebox plugin.")
    endif()
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set(SV_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/sdk" CACHE PATH "Extracted SDK root, or tools/sdk in the repository")
    if(EXISTS "${SV_SDK_ROOT}/tools/sv_plugin.cmake")
        include("${SV_SDK_ROOT}/tools/sv_plugin.cmake")
    else()
        include("${SV_SDK_ROOT}/sv_plugin.cmake")
    endif()
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/instruments.ini" JB_DEFAULT_TEXT)
string(REPLACE "\\" "\\\\" JB_DEFAULT_TEXT "${JB_DEFAULT_TEXT}")
string(REPLACE "\"" "\\\"" JB_DEFAULT_TEXT "${JB_DEFAULT_TEXT}")
string(REPLACE "\r" "" JB_DEFAULT_TEXT "${JB_DEFAULT_TEXT}")
string(REPLACE "\n" "\\n\"\n\"" JB_DEFAULT_TEXT "${JB_DEFAULT_TEXT}")
configure_file(defaults.h.in jukebox_defaults.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/instruments.ini")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sounds.ini" JB_SOUND_TEXT)
string(REPLACE "\\" "\\\\" JB_SOUND_TEXT "${JB_SOUND_TEXT}")
string(REPLACE "\"" "\\\"" JB_SOUND_TEXT "${JB_SOUND_TEXT}")
string(REPLACE "\r" "" JB_SOUND_TEXT "${JB_SOUND_TEXT}")
string(REPLACE "\n" "\\n\"\n\"" JB_SOUND_TEXT "${JB_SOUND_TEXT}")
configure_file(sounds.h.in jukebox_sounds.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/sounds.ini")
sv_add_plugin(sv-jukebox ID sv.jukebox MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/plugin.ini"
    SOURCES jukebox.c midi.c instruments.c files.c)
target_include_directories(sv-jukebox PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")

# Test programs. The Python tests in tests/ run them. They sit in tests/ of the build folder.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tests")
function(jukebox_test_program name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SV_PLUGIN_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${name} PRIVATE /W4 /WX /utf-8)
endfunction()
jukebox_test_program(jukebox-instruments-test tests/jukebox_instruments_test.c instruments.c)
jukebox_test_program(jukebox-midi-test tests/jukebox_midi_driver.c midi.c)
jukebox_test_program(jukebox-host-test tests/jukebox_host_test.c)
add_dependencies(jukebox-host-test sv-jukebox)
